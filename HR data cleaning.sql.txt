select *,
row_number() over(partition by id,first_name,last_name,birthdate,gender,race,
department,jobtitle,location,hire_date,location_city,location_state
order by id) as row_num
from human_resources; 

with duplicate_cte as(
select *,
row_number() over(partition by id, first_name, last_name, birthdate, gender, race,
department, jobtitle, location, hire_date, location_city, location_state
order by id) as row_num
from human_resources)
select * from duplicate_cte
where row_num >1; -- it is showing zero/nothing means no duplicate data.

/* If duplicates had been present, you would use that same CTE structure and replace SELECT * with DELETE to remove them instantly while keeping one original copy.

Here is how the exact DELETE statement works:

WITH duplicate_cte AS (
    SELECT id,
        ROW_NUMBER() OVER(
            PARTITION BY id, first_name, last_name, birthdate, gender, race,
                         department, jobtitle, location, hire_date, location_city, location_state
            ORDER BY id
        ) AS row_num
    FROM human_resources
)
DELETE FROM human_resources
WHERE id IN (
    SELECT id 
    FROM duplicate_cte 
    WHERE row_num > 1
); 

How this works:
row_num = 1: The first occurrence of the record is kept safe.

row_num > 1: Every extra copy (2nd, 3rd duplicate, etc.) is identified by the subquery 
and deleted from human_resources.    */

-- what is age distribution of employees in the company

select 
case
when age >=18 and age <=24 then '18-24'
when age >=25 and age <=34 then '25-34'
when age >=35 and age <=45 then '35-45'
when age >=36 and age <=46 then '36-46'

--which department has highest turnover rate

select department,
 total_count,
 terminated_count,
 round(terminated_count*1.0/total_count,2) as termination_rate
 from(
 select department,
 count(*) as total_count,
 sum(case when termdate is not null and termdate <= current_date then 1 else 0 end) as terminated_count
 from human_resources where age >= 18
 group by department) as subquery
 order by termination_rate desc;

-- what is distribution of employees across location city and state

select location_city, location_state, count(*) as count
from human_resources hr 
where age>=18 and termdate is null 
group by hr.location_city,hr.location_state 
order by count desc;

-- How has teh company's employee count changed over time based on hire and term dates

select 
year,
hires,
terminations,
hires - terminations as net_change,
round((hires - terminations)*100/hires,2) as net_change_percent
from(
select
extract (year from hire_date) as year,
count(*) as hires,
sum(case when termdate is not null and termdate <= current_date then 1 else 0 end) as terminations
from human_resources
where age >= 18
group by year
) as sub_query
order by year asc;
